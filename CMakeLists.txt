cmake_minimum_required(VERSION 3.16)
project(SchematicEditor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base propgrid xml)
include(${wxWidgets_USE_FILE})
set(MODEL_SOURCES src/NetModel.cpp src/SchematicIO.cpp src/ComponentCatalog.cpp src/CircuitTools.cpp src/DigitalSimulator.cpp)
add_executable(SchematicEditor src/minimal.cpp src/MainFrame.cpp src/SchematicCanvas.cpp src/LibraryPanel.cpp src/PropertyPanel.cpp src/TextReportDialog.cpp ${MODEL_SOURCES})
target_link_libraries(SchematicEditor PRIVATE ${wxWidgets_LIBRARIES})
enable_testing()
add_executable(model_tests tests/model_tests.cpp ${MODEL_SOURCES})
target_include_directories(model_tests PRIVATE src)
target_link_libraries(model_tests PRIVATE ${wxWidgets_LIBRARIES})
add_test(NAME model_tests COMMAND model_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples/sample-schematic.ses)
add_executable(dynamic_tests tests/dynamic_tests.cpp ${MODEL_SOURCES} src/KiCadExport.cpp)
target_include_directories(dynamic_tests PRIVATE src)
target_link_libraries(dynamic_tests PRIVATE ${wxWidgets_LIBRARIES})
add_test(NAME dynamic_tests COMMAND dynamic_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples)
