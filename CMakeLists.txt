cmake_minimum_required(VERSION 3.16)
project(SchematicEditor VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base propgrid xml)
include(${wxWidgets_USE_FILE})
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SOURCE_REV OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT SOURCE_REV)
  set(SOURCE_REV "unversioned")
endif()
add_compile_definitions(SE_SOURCE_REV="${SOURCE_REV}")
set(MODEL_SOURCES src/NetModel.cpp src/SchematicIO.cpp src/ComponentCatalog.cpp src/CircuitTools.cpp src/DigitalSimulator.cpp src/KiCadExport.cpp src/EditingTools.cpp)
set(GUI_SOURCES src/MainFrame.cpp src/SchematicCanvas.cpp src/LibraryPanel.cpp src/PropertyPanel.cpp src/TextReportDialog.cpp)
add_library(schematic_model STATIC ${MODEL_SOURCES})
target_include_directories(schematic_model PUBLIC src)
target_link_libraries(schematic_model PUBLIC ${wxWidgets_LIBRARIES})
add_executable(SchematicEditor src/minimal.cpp ${GUI_SOURCES})
target_link_libraries(SchematicEditor PRIVATE schematic_model)
add_executable(schematic-cli src/cli.cpp)
target_link_libraries(schematic-cli PRIVATE schematic_model)
enable_testing()
foreach(name model dynamic editing)
  add_executable(${name}_tests tests/${name}_tests.cpp)
  target_link_libraries(${name}_tests PRIVATE schematic_model)
endforeach()
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/fixtures ${CMAKE_BINARY_DIR}/gui-artifacts)
add_test(NAME model_tests COMMAND model_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples/sample-schematic.ses)
add_test(NAME dynamic_tests COMMAND dynamic_tests ${CMAKE_BINARY_DIR}/fixtures ${CMAKE_CURRENT_SOURCE_DIR}/examples/sample-library.xml)
add_test(NAME editing_tests COMMAND editing_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_integration.py $<TARGET_FILE:schematic-cli> ${CMAKE_CURRENT_SOURCE_DIR}/examples)
endif()
add_executable(gui_widget_tests tests/gui_widget_tests.cpp ${GUI_SOURCES})
target_link_libraries(gui_widget_tests PRIVATE schematic_model)
add_test(NAME gui_widget_tests COMMAND gui_widget_tests ${CMAKE_CURRENT_SOURCE_DIR}/examples ${CMAKE_BINARY_DIR}/gui-artifacts)
set_tests_properties(gui_widget_tests PROPERTIES SKIP_RETURN_CODE 77 LABELS gui)
install(TARGETS SchematicEditor schematic-cli RUNTIME DESTINATION bin)
install(DIRECTORY examples/ DESTINATION share/SchematicEditor/examples)
